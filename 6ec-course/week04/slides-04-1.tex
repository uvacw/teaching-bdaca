	\def\chapterprogress{0}%

		\fill[fg!20!bg] (0,-1) rectangle (1,1);
		\fill (0,-1) rectangle (\chapterprogress,1);
	\end{tikzpicture}%
}

\begin{frame}{Document embeddings}
Do these headlines describe a similar event?

\begin{quote}
	City raises bus fares\\
	Public transport gets more expensive
\end{quote}

\begin{block}{Bag-of-words}
	The headlines share no words, so their count vectors have zero cosine similarity.
\end{block}

\begin{block}{Document embeddings}
	A pretrained model turns each headline into a vector. Related texts can have similar vectors even when they use different words.
\end{block}

Next week, BERTopic uses embeddings to group texts and a variant of TF--IDF to describe each group.
\end{frame}
